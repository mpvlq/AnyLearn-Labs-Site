function project(){

    n.pz=z2;n.ps=CAM/(CAM+z2);n.px=x1*n.ps;n.py=y2*n.ps;});
}

const calm = !!(window.matchMedia&&matchMedia('(prefers-reduced-motion: reduce)').matches);

function fireNode(n,gen){
  n.fire=1; if(gen>=3) return;
  const es=vedges.filter(e=>e.s===n||e.t===n).sort(()=>Math.random()-.5), k=Math.min(es.length,1+Math.floor(Math.random()*3));
  for(let i=0;i<k;i++){const e=es[i]; pulses.push({e,from:n,to:e.s===n?e.t:e.s,t:0,dur:650+Math.random()*550,gen});}
}
function stepFx(now){
  const dt=Math.min(60,now-lastT); lastT=now;
  if(!calm && now>=nextFire && vis.length){ fireNode(vis[Math.floor(Math.random()*vis.length)],0); nextFire=now+450+Math.random()*1200; }
  nodes.forEach(n=>{ if(n.fire>0) n.fire=Math.max(0,n.fire-dt/950); });
  pulses=pulses.filter(p=>{ p.t+=dt/p.dur; if(p.t>=1){ if(vis.includes(p.to)&&Math.random()<.72-.2*p.gen) fireNode(p.to,p.gen+1); return false; } return vedges.includes(p.e); });
}
